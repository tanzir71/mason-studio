## Goal

`tokensToCss(set)` writes the `:root` variables plus tablet and mobile media blocks.

## Context

Engine module, tokens area. Pages link tokens.css before styles.css, so variables are available everywhere. The fixture's tokens.css is exactly the expected output for its tokens.json. Pure functions: no I/O. Token names use dots (`color.ink`); CSS variables use dashes (`--color-ink`).

## Contracts (do not modify)

```ts
// src/contracts/engine.ts
/**
 * tokensToCss(set): "/* Generated by Mason from tokens.json. Do not edit. *\/\n:root {\n  --a: v;\n}\n" (aliases →
 * `var(--b)`), then for "tablet" then "mobile" when any token has a byBreakpoint value:
 * "@media (max-width: 768px) {\n  :root {\n    --a: v;\n  }\n}\n" (375 for mobile).
 */
export type TokensToCss = (set: TokenSet) => string;
```

## Steps

1. Create `src/engine/tokens/to-css.ts` with exactly this content:

```ts
import type { TokenSet } from "../../contracts/model.ts";
import { cssVarName } from "./dtcg.ts";

/** See TokensToCss in src/contracts/engine.ts. */
export function tokensToCss(set: TokenSet): string {
  const tokens = [...set.tokens].sort((a, b) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0));
  const lines = ["/* Generated by Mason from tokens.json. Do not edit. */", ":root {"];
  for (const t of tokens) lines.push(`  ${cssVarName(t.name)}: ${t.alias ? `var(${cssVarName(t.alias)})` : t.value};`);
  lines.push("}");
  for (const [bp, max] of [["tablet", 768], ["mobile", 375]] as const) {
    const list = tokens.filter((t) => t.byBreakpoint[bp] !== undefined);
    if (list.length === 0) continue;
    lines.push(`@media (max-width: ${max}px) {`, "  :root {");
    for (const t of list) lines.push(`    ${cssVarName(t.name)}: ${t.byBreakpoint[bp]};`);
    lines.push("  }", "}");
  }
  return lines.join("\n") + "\n";
}
```

```ts
      { name: "a", type: "color", value: "#000000", alias: null, byBreakpoint: {} },
    ] });
    expect(css).toBe("/* Generated by Mason from tokens.json. Do not edit. */\n:root {\n  --a: #000000;\n  --b: var(--a);\n}\n");
```
